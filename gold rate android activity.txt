package com.example.goldpricecalculator;

import androidx.appcompat.app.AppCompatActivity;
import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;
import android.widget.Toast;

public class MainActivity extends AppCompatActivity {

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        // Bind UI elements
        EditText editTextGoldRate = findViewById(R.id.editTextGoldRate);
        EditText editTextCharge = findViewById(R.id.editTextCharge);
        Button buttonCalculate = findViewById(R.id.buttonCalculate);
        TextView textViewGoldPrice = findViewById(R.id.textViewGoldPrice);
        TextView textViewPayment = findViewById(R.id.textViewPayment);

        // Set click listener for the button
        buttonCalculate.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                // Get inputs from EditText fields
                String rateInput = editTextGoldRate.getText().toString();
                String chargeInput = editTextCharge.getText().toString();

                // Check if the input fields are not empty
                if (!rateInput.isEmpty() && !chargeInput.isEmpty()) {
                    // Parse input to double
                    double rate = Double.parseDouble(rateInput);
                    double charge = Double.parseDouble(chargeInput);

                    // Constants and calculation
                    double p = 2.23;
                    double r = rate / 11.664 * p;
                    double c = r - charge;

                    // Set formatted results to the TextViews
                    textViewGoldPrice.setText(String.format("Gold Price is = %.2f", r));
                    textViewPayment.setText(String.format("Your Payment is = %.2f", c));
                } else {
                    // Show a Toast if the input fields are empty
                    Toast.makeText(MainActivity.this, "Please enter values for both fields", Toast.LENGTH_SHORT).show();
                }
            }
        });
    }
}
